async function generatePix(event) { event.preventDefault(); const name = document.getElementById('buyerName').value; const email = document.getElementById('buyerEmail').value; const cpf = document.getElementById('buyerCpf').value.replace(/\D/g, ''); document.getElementById('checkoutFormStage').classList.add('hidden'); document.getElementById('checkoutLoadingStage').classList.remove('hidden'); try { // Chama a nossa rota interna na pasta /api/gerar-pix const response = await fetch('/api/gerar-pix', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ amount: currentPrice, description: currentProduct, payer: { name: name, email: email, taxId: cpf } }) }); const pixData = await response.json(); if (pixData.qrCode || pixData.pixCopiaECola || pixData.payload) { const qrCodeImg = pixData.qrCodeBase64 || pixData.qrCode || `https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=${encodeURIComponent(pixData.pixCopiaECola || pixData.payload)}`; const copyCode = pixData.pixCopiaECola || pixData.payload; const txId = pixData.id || pixData.transactionId; document.getElementById('pixQrCodeImg').src = qrCodeImg; document.getElementById('pixCopyInput').value = copyCode; document.getElementById('checkoutLoadingStage').classList.add('hidden'); document.getElementById('checkoutPixStage').classList.remove('hidden'); // Simula a aprovação automática após o pagamento real ser detectado ou para testes checkInterval = setTimeout(() => { showSuccessStage(); }, 20000); } else { alert('Erro ao gerar Pix: ' + (pixData.error || 'Tente novamente.')); document.getElementById('checkoutLoadingStage').classList.add('hidden'); document.getElementById('checkoutFormStage').classList.remove('hidden'); } } catch (err) { console.error(err); alert('Erro de conexão ao gerar o Pix.'); document.getElementById('checkoutLoadingStage').classList.add('hidden'); document.getElementById('checkoutFormStage').classList.remove('hidden'); } }
